SELECT h.hacker_id, h.name, COUNT(c.challenge_id) AS cnt
FROM Hackers h
JOIN Challenges c ON h.hacker_id = c.hacker_id
GROUP BY h.hacker_id, h.name
HAVING cnt = (
    SELECT MAX(cnt) FROM (
        SELECT COUNT(*) AS cnt FROM Challenges GROUP BY hacker_id
    ) max_t
)
OR cnt NOT IN (
    SELECT cnt FROM (
        SELECT COUNT(*) AS cnt FROM Challenges GROUP BY hacker_id
    ) dup_t
    GROUP BY cnt
    HAVING COUNT(*) > 1
)
ORDER BY cnt DESC, h.hacker_id;
