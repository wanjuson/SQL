SELECT GROUP_CONCAT(num SEPARATOR ' ')
FROM (
    SELECT a.N AS num
    FROM (
        SELECT (a.N + b.N * 10 + c.N * 100 + 2) AS N
        FROM
          (SELECT 0 AS N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4
           UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) a,
          (SELECT 0 AS N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4
           UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) b,
          (SELECT 0 AS N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4
           UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) c
    ) a
    WHERE a.N BETWEEN 2 AND 1000
) nums
WHERE NOT EXISTS (
    SELECT 1
    FROM (
        SELECT (x.N + y.N * 10 + 2) AS d
        FROM
          (SELECT 0 AS N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4
           UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) x,
          (SELECT 0 AS N UNION SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4
           UNION SELECT 5 UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9) y
    ) divisors
    WHERE divisors.d BETWEEN 2 AND SQRT(nums.num)
      AND nums.num % divisors.d = 0
)
GROUP BY 1;
