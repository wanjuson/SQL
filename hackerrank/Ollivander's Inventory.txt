SELECT w.id, w.age, w.coins_needed, w.power
FROM Wands w
JOIN Wands_Property p ON w.code = p.code
WHERE p.is_evil = 0
  AND w.coins_needed = (
      SELECT MIN(w2.coins_needed)
      FROM Wands w2
      JOIN Wands_Property p2 ON w2.code = p2.code
      WHERE p2.is_evil = 0
        AND w2.power = w.power
        AND w2.age = w.age
  )
ORDER BY w.power DESC, w.age DESC;
